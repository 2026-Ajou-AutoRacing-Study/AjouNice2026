# Field terminal logging ------------------------------------
# Start one shared session in the roscore terminal, then pipe every long-running
# command through field_log_tee in its own terminal.  The pointer file lets newly
# opened terminals join the same session without copying a timestamp by hand.
export AJOUNICE_FIELD_LOG_POINTER="${AJOUNICE_FIELD_LOG_POINTER:-/tmp/ajounice_field_log_prefix}"

field_log_new() {
  local log_dir tag prefix index pointer_tmp

  log_dir="${AJOUNICE_FIELD_LOG_DIR:-$AJOUNICE_ROOT/artifacts/field_logs}"
  mkdir -p -- "$log_dir" || return

  tag="$(date +%y%m%d_%H%M%S)"
  prefix="$log_dir/$tag"
  index=1
  while compgen -G "${prefix}_*" >/dev/null; do
    printf -v prefix '%s/%s_%02d' "$log_dir" "$tag" "$index"
    ((index++))
  done

  pointer_tmp="${AJOUNICE_FIELD_LOG_POINTER}.tmp.$$"
  printf '%s\n' "$prefix" >"$pointer_tmp" || return
  chmod 600 "$pointer_tmp" 2>/dev/null || true
  mv -f -- "$pointer_tmp" "$AJOUNICE_FIELD_LOG_POINTER" || return
  export AJOUNICE_FIELD_LOG_PREFIX="$prefix"

  {
    echo "started_at=$(date --iso-8601=seconds)"
    echo "hostname=$(hostname)"
    echo "repo_root=$AJOUNICE_ROOT"
    echo "git_branch=$(git -C "$AJOUNICE_ROOT" branch --show-current 2>/dev/null || true)"
    echo "git_sha=$(git -C "$AJOUNICE_ROOT" rev-parse HEAD 2>/dev/null || true)"
    echo "ros_master_uri=${ROS_MASTER_URI:-unset}"
    echo "ros_ip=${ROS_IP:-unset}"
  } >"${prefix}_session.txt"

  echo "Field log prefix: $prefix"
  echo "Session metadata: ${prefix}_session.txt"
}

field_log_prefix() {
  local prefix log_dir

  prefix=""
  if [[ -r "$AJOUNICE_FIELD_LOG_POINTER" ]]; then
    IFS= read -r prefix <"$AJOUNICE_FIELD_LOG_POINTER"
  else
    prefix="${AJOUNICE_FIELD_LOG_PREFIX:-}"
  fi

  log_dir="${AJOUNICE_FIELD_LOG_DIR:-$AJOUNICE_ROOT/artifacts/field_logs}"
  if [[ -z "$prefix" || "$prefix" != "$log_dir"/* ]]; then
    echo "ERROR: no valid field log session; run field_log_new in Terminal 1 first" >&2
    return 1
  fi

  printf '%s\n' "$prefix"
}

field_log_tee() {
  local label prefix started log_file index

  if [[ $# -ne 1 || ! "$1" =~ ^[A-Za-z0-9][A-Za-z0-9_-]*$ ]]; then
    echo "usage: field_log_tee LOG_LABEL" >&2
    return 2
  fi
  label="$1"
  prefix="$(field_log_prefix)" || return
  started="$(date +%y%m%d_%H%M%S)"
  log_file="${prefix}_${label}_${started}.log"
  index=1
  while [[ -e "$log_file" ]]; do
    printf -v log_file '%s_%s_%s_%02d.log' \
      "$prefix" "$label" "$started" "$index"
    ((index++))
  done

  echo "Field log file: $log_file" >&2
  tee -- "$log_file"
}

field_log_list() {
  local prefix log_dir base

  prefix="$(field_log_prefix)" || return
  log_dir="$(dirname -- "$prefix")"
  base="$(basename -- "$prefix")"
  echo "Current field log prefix: $prefix"
  find "$log_dir" -maxdepth 1 -type f -name "${base}_*" -exec ls -lh -- {} +
  if [[ -d "${prefix}_console" ]]; then
    find "${prefix}_console" -maxdepth 1 -type f -exec ls -lh -- {} +
  fi
}
