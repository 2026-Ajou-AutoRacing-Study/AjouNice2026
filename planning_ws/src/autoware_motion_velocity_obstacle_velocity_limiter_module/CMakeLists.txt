cmake_minimum_required(VERSION 3.0.2)
project(autoware_motion_velocity_obstacle_velocity_limiter_module)
find_package(catkin REQUIRED COMPONENTS autoware_map_msgs autoware_perception_msgs autoware_planning_msgs geometry_msgs nav_msgs pluginlib roscpp sensor_msgs tf2 visualization_msgs)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
find_package(PkgConfig REQUIRED)
pkg_check_modules(OPENCV4 REQUIRED opencv4)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/AW149Providers.cmake")
aw149_provider(GRID_MAP_UTILS autoware_grid_map_utils autoware_grid_map_utils)
aw149_provider(MOTION_UTILS autoware_motion_utils autoware_motion_utils)
aw149_provider(MOTION_VELOCITY_PLANNER_COMMON autoware_motion_velocity_planner_common autoware_motion_velocity_planner_common)
aw149_provider(ROUTE_HANDLER autoware_route_handler autoware_route_handler)
aw149_provider(UNIVERSE_UTILS autoware_universe_utils NONE)
aw149_provider(VEHICLE_INFO_UTILS autoware_vehicle_info_utils vehicle_info_utils)
set(UNIVERSE_UTILS_INCLUDE_DIR "" CACHE PATH "installed universe-utils include directory")
set(MOTION_VELOCITY_PLANNER_COMMON_INCLUDE_DIR "" CACHE PATH "installed motion-velocity-planner-common include directory")
set(MOTION_VELOCITY_PLANNER_COMMON_LIBRARY "" CACHE FILEPATH "installed motion-velocity-planner-common library")
set(MOTION_UTILS_INCLUDE_DIR "" CACHE PATH "installed motion-utils include directory")
set(MOTION_UTILS_LIBRARY "" CACHE FILEPATH "installed motion-utils library")
set(ROUTE_HANDLER_INCLUDE_DIR "" CACHE PATH "installed route-handler include directory")
set(ROUTE_HANDLER_LIBRARY "" CACHE FILEPATH "installed route-handler library")
set(VEHICLE_INFO_UTILS_INCLUDE_DIR "" CACHE PATH "installed vehicle-info-utils include directory")
set(VEHICLE_INFO_UTILS_LIBRARY "" CACHE FILEPATH "installed vehicle-info-utils library")
set(GRID_MAP_UTILS_INCLUDE_DIR "" CACHE PATH "installed grid-map-utils include directory")
set(GRID_MAP_UTILS_LIBRARY "" CACHE FILEPATH "installed grid-map-utils library")
find_library(GRID_MAP_CORE_LIBRARY NAMES grid_map_core REQUIRED)
find_library(GRID_MAP_ROS_LIBRARY NAMES grid_map_ros REQUIRED)
if(NOT EXISTS "${UNIVERSE_UTILS_INCLUDE_DIR}")
  message(FATAL_ERROR "Set UNIVERSE_UTILS_INCLUDE_DIR to its staged ROS1 provider")
endif()
if(NOT EXISTS "${MOTION_VELOCITY_PLANNER_COMMON_INCLUDE_DIR}" OR NOT EXISTS "${MOTION_VELOCITY_PLANNER_COMMON_LIBRARY}")
  message(FATAL_ERROR "Set MOTION_VELOCITY_PLANNER_COMMON_INCLUDE_DIR and _LIBRARY to staged ROS1 providers")
endif()
if(NOT EXISTS "${MOTION_UTILS_INCLUDE_DIR}" OR NOT EXISTS "${MOTION_UTILS_LIBRARY}")
  message(FATAL_ERROR "Set MOTION_UTILS_INCLUDE_DIR and _LIBRARY to staged ROS1 providers")
endif()
if(NOT EXISTS "${ROUTE_HANDLER_INCLUDE_DIR}" OR NOT EXISTS "${ROUTE_HANDLER_LIBRARY}")
  message(FATAL_ERROR "Set ROUTE_HANDLER_INCLUDE_DIR and _LIBRARY to staged ROS1 providers")
endif()
if(NOT EXISTS "${VEHICLE_INFO_UTILS_INCLUDE_DIR}" OR NOT EXISTS "${VEHICLE_INFO_UTILS_LIBRARY}")
  message(FATAL_ERROR "Set VEHICLE_INFO_UTILS_INCLUDE_DIR and _LIBRARY to staged ROS1 providers")
endif()
if(NOT EXISTS "${GRID_MAP_UTILS_INCLUDE_DIR}" OR NOT EXISTS "${GRID_MAP_UTILS_LIBRARY}")
  message(FATAL_ERROR "Set GRID_MAP_UTILS_INCLUDE_DIR and _LIBRARY to staged ROS1 providers")
endif()
file(GLOB MODULE_SOURCES src/*.cpp)
catkin_package(LIBRARIES ${PROJECT_NAME} CATKIN_DEPENDS autoware_map_msgs autoware_perception_msgs autoware_planning_msgs geometry_msgs nav_msgs pluginlib roscpp sensor_msgs tf2 visualization_msgs)
include_directories(src ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${OPENCV4_INCLUDE_DIRS} ${UNIVERSE_UTILS_INCLUDE_DIR} ${MOTION_VELOCITY_PLANNER_COMMON_INCLUDE_DIR} ${MOTION_UTILS_INCLUDE_DIR} ${ROUTE_HANDLER_INCLUDE_DIR} ${VEHICLE_INFO_UTILS_INCLUDE_DIR} ${GRID_MAP_UTILS_INCLUDE_DIR})
add_library(${PROJECT_NAME} SHARED ${MODULE_SOURCES})
target_link_libraries(${PROJECT_NAME} ${catkin_LIBRARIES} ${PCL_LIBRARIES} ${OPENCV4_LIBRARIES} ${MOTION_VELOCITY_PLANNER_COMMON_LIBRARY} ${MOTION_UTILS_LIBRARY} ${ROUTE_HANDLER_LIBRARY} ${VEHICLE_INFO_UTILS_LIBRARY} ${GRID_MAP_UTILS_LIBRARY} ${GRID_MAP_ROS_LIBRARY} ${GRID_MAP_CORE_LIBRARY})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
add_executable(verify_obstacle_velocity_limiter_plugin test/pluginlib_r025.cpp)
target_link_libraries(verify_obstacle_velocity_limiter_plugin ${catkin_LIBRARIES})
target_compile_features(verify_obstacle_velocity_limiter_plugin PRIVATE cxx_std_17)
add_executable(verify_obstacle_velocity_limiter_init test/plugin_init_r025.cpp)
target_link_libraries(verify_obstacle_velocity_limiter_init ${catkin_LIBRARIES})
target_compile_features(verify_obstacle_velocity_limiter_init PRIVATE cxx_std_17)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES package.xml plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY config/ DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/config)
