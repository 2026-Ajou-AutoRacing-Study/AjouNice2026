cmake_minimum_required(VERSION 3.0.2)
project(autoware_behavior_velocity_speed_bump_module)
find_package(catkin REQUIRED COMPONENTS geometry_msgs pluginlib roscpp visualization_msgs)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
find_package(lanelet2_core REQUIRED)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/AW149Providers.cmake")
aw149_provider(BVPC_COMMON autoware_behavior_velocity_planner_common autoware_behavior_velocity_planner_common)
aw149_provider(BVP_MANAGER autoware_behavior_velocity_planner autoware_behavior_velocity_planner_manager)
aw149_provider(LANELET2_UTILS autoware_lanelet2_utils autoware_lanelet2_utils)
aw149_provider(LANELET_EXTENSION autoware_lanelet2_extension autoware_lanelet2_extension_lib)
aw149_provider(MAP_MSGS autoware_map_msgs NONE)
aw149_provider(MOTION_UTILS autoware_motion_utils autoware_motion_utils)
aw149_provider(OBJECTS autoware_objects_of_interest_marker_interface autoware_objects_of_interest_marker_interface)
aw149_provider(OSQP osqp osqp)
aw149_provider(OSQP_INTERFACE autoware_osqp_interface autoware_osqp_interface)
aw149_provider(QP_INTERFACE autoware_qp_interface autoware_qp_interface)
aw149_provider(ROUTE_HANDLER autoware_route_handler autoware_route_handler)
aw149_provider(TRAJECTORY autoware_trajectory autoware_trajectory)
aw149_provider(UTILS autoware_universe_utils NONE)
aw149_provider(UTILS_DEBUG autoware_utils_debug autoware_utils_debug)
aw149_provider(UTILS_GEOMETRY autoware_utils_geometry autoware_utils_geometry)
aw149_provider(UTILS_VISUALIZATION autoware_utils_visualization autoware_utils_visualization)
aw149_provider(VEHICLE_INFO autoware_vehicle_info_utils vehicle_info_utils)
aw149_provider(VELOCITY_SMOOTHER autoware_velocity_smoother autoware_velocity_smoother)
set(BVPC_COMMON_INCLUDE_DIR "" CACHE PATH "staged behavior-velocity common include")
set(BVPC_COMMON_LIBRARY "" CACHE FILEPATH "staged behavior-velocity common library")
set(BVP_MANAGER_LIBRARY "" CACHE FILEPATH "staged BVP manager runtime closure")
set(TRAJECTORY_INCLUDE_DIR "" CACHE PATH "staged trajectory include")
set(TRAJECTORY_LIBRARY "" CACHE FILEPATH "staged trajectory library")
set(ROUTE_HANDLER_INCLUDE_DIR "" CACHE PATH "staged route-handler include")
set(ROUTE_HANDLER_LIBRARY "" CACHE FILEPATH "staged route-handler library")
set(MOTION_UTILS_INCLUDE_DIR "" CACHE PATH "staged motion-utils include")
set(MOTION_UTILS_LIBRARY "" CACHE FILEPATH "staged motion-utils library")
set(LANELET_EXTENSION_INCLUDE_DIR "" CACHE PATH "staged lanelet extension include")
set(LANELET_EXTENSION_LIBRARY "" CACHE FILEPATH "staged lanelet extension library")
set(UTILS_INCLUDE_DIR "" CACHE PATH "staged universe-utils include")
set(UTILS_DEBUG_LIBRARY "" CACHE FILEPATH "staged universe-utils debug library")
set(UTILS_GEOMETRY_LIBRARY "" CACHE FILEPATH "staged universe-utils geometry library")
set(UTILS_VISUALIZATION_LIBRARY "" CACHE FILEPATH "staged universe-utils visualization library")
set(VEHICLE_INFO_INCLUDE_DIR "" CACHE PATH "staged vehicle-info include")
set(VEHICLE_INFO_LIBRARY "" CACHE FILEPATH "staged vehicle-info library")
set(OBJECTS_INCLUDE_DIR "" CACHE PATH "staged objects-of-interest include")
set(OBJECTS_LIBRARY "" CACHE FILEPATH "staged objects-of-interest library")
set(MAP_MSGS_INCLUDE_DIR "" CACHE PATH "staged map messages include")
set(VELOCITY_SMOOTHER_LIBRARY "" CACHE FILEPATH "staged velocity-smoother library")
set(LANELET2_UTILS_LIBRARY "" CACHE FILEPATH "staged lanelet2-utils library")
set(QP_INTERFACE_LIBRARY "" CACHE FILEPATH "staged qp-interface library")
set(OSQP_INTERFACE_LIBRARY "" CACHE FILEPATH "staged osqp-interface library")
set(OSQP_LIBRARY "" CACHE FILEPATH "staged OSQP provider library")
foreach(path IN ITEMS BVPC_COMMON_INCLUDE_DIR BVPC_COMMON_LIBRARY BVP_MANAGER_LIBRARY TRAJECTORY_INCLUDE_DIR TRAJECTORY_LIBRARY ROUTE_HANDLER_INCLUDE_DIR ROUTE_HANDLER_LIBRARY MOTION_UTILS_INCLUDE_DIR MOTION_UTILS_LIBRARY LANELET_EXTENSION_INCLUDE_DIR LANELET_EXTENSION_LIBRARY UTILS_INCLUDE_DIR UTILS_DEBUG_LIBRARY VEHICLE_INFO_INCLUDE_DIR VEHICLE_INFO_LIBRARY OBJECTS_INCLUDE_DIR OBJECTS_LIBRARY MAP_MSGS_INCLUDE_DIR)
  if(NOT EXISTS "${${path}}")
    message(FATAL_ERROR "Set ${path} to its staged ROS1 provider")
  endif()
endforeach()
set(SPEED_BUMP_RUNTIME_LIBRARIES
  "${VELOCITY_SMOOTHER_LIBRARY}" "${MOTION_UTILS_LIBRARY}"
  "${UTILS_DEBUG_LIBRARY}" "${UTILS_GEOMETRY_LIBRARY}" "${UTILS_VISUALIZATION_LIBRARY}"
  "${LANELET2_UTILS_LIBRARY}" "${LANELET_EXTENSION_LIBRARY}" "${VEHICLE_INFO_LIBRARY}"
  "${QP_INTERFACE_LIBRARY}" "${OSQP_INTERFACE_LIBRARY}" "${OSQP_LIBRARY}")
foreach(path IN LISTS SPEED_BUMP_RUNTIME_LIBRARIES)
  if(NOT EXISTS "${path}")
    message(FATAL_ERROR "Set the staged ROS1 runtime closure library: ${path}")
  endif()
endforeach()
catkin_package(LIBRARIES ${PROJECT_NAME} CATKIN_DEPENDS geometry_msgs pluginlib roscpp visualization_msgs)
include_directories(src ${catkin_INCLUDE_DIRS} ${BVPC_COMMON_INCLUDE_DIR} ${TRAJECTORY_INCLUDE_DIR} ${ROUTE_HANDLER_INCLUDE_DIR} ${MOTION_UTILS_INCLUDE_DIR} ${LANELET_EXTENSION_INCLUDE_DIR} ${UTILS_INCLUDE_DIR} ${VEHICLE_INFO_INCLUDE_DIR} ${OBJECTS_INCLUDE_DIR} ${MAP_MSGS_INCLUDE_DIR} ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
add_library(${PROJECT_NAME} SHARED src/manager.cpp src/scene.cpp)
target_link_libraries(${PROJECT_NAME} "${BVPC_COMMON_LIBRARY}" "${BVP_MANAGER_LIBRARY}" "${TRAJECTORY_LIBRARY}" "${ROUTE_HANDLER_LIBRARY}" "${MOTION_UTILS_LIBRARY}" "${LANELET_EXTENSION_LIBRARY}" "${UTILS_DEBUG_LIBRARY}" "${VEHICLE_INFO_LIBRARY}" "${OBJECTS_LIBRARY}" ${SPEED_BUMP_RUNTIME_LIBRARIES} lanelet2_core::lanelet2_core ${catkin_LIBRARIES})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
add_library(${PROJECT_NAME}_normal SHARED
  src/normal/manager.cpp src/normal/scene.cpp src/normal/debug.cpp src/util.cpp)
target_link_libraries(${PROJECT_NAME}_normal "${BVPC_COMMON_LIBRARY}" "${BVP_MANAGER_LIBRARY}" "${TRAJECTORY_LIBRARY}" "${ROUTE_HANDLER_LIBRARY}" "${MOTION_UTILS_LIBRARY}" "${LANELET_EXTENSION_LIBRARY}" "${UTILS_DEBUG_LIBRARY}" "${VEHICLE_INFO_LIBRARY}" "${OBJECTS_LIBRARY}" ${SPEED_BUMP_RUNTIME_LIBRARIES} lanelet2_core::lanelet2_core ${catkin_LIBRARIES})
target_compile_features(${PROJECT_NAME}_normal PUBLIC cxx_std_17)
install(TARGETS ${PROJECT_NAME} ${PROJECT_NAME}_normal LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES package.xml plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})

  catkin_add_gtest(test_speed_bump_plugin test/test_speed_bump_plugin.cpp)
  if(TARGET test_speed_bump_plugin)
    target_link_libraries(test_speed_bump_plugin ${PROJECT_NAME} ${PROJECT_NAME}_normal ${SPEED_BUMP_RUNTIME_LIBRARIES} ${catkin_LIBRARIES})
  endif()
  catkin_add_gtest(test_speed_bump_scene test/test_speed_bump_scene.cpp)
  if(TARGET test_speed_bump_scene)
    target_link_libraries(test_speed_bump_scene ${PROJECT_NAME} ${PROJECT_NAME}_normal ${SPEED_BUMP_RUNTIME_LIBRARIES} ${catkin_LIBRARIES})
  endif()
endif()
