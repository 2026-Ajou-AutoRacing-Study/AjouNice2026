  foreach(directory IN LISTS DYNAMIC_OBSTACLE_AVOIDANCE_RUNTIME_LIBRARY_DIRS)
    if(NOT IS_DIRECTORY "${directory}")
      message(FATAL_ERROR "Dynamic Obstacle Avoidance runtime library directory does not exist: ${directory}")
    endif()
    # The DOA candidate is a shared provider with an explicit transitive
    # closure.  Apply that closure to every executable created below (the core
    # regression tests included), not only the node binary; otherwise a test
    # can link against the provider while the linker cannot resolve its DSO
    # dependencies.
    add_link_options("-Wl,-rpath-link,${directory}")
    target_link_options(behavior_path_planner_node PRIVATE "-Wl,-rpath-link,${directory}")
    target_link_options(aw149_bpp_doa_slot_init PRIVATE "-Wl,-rpath-link,${directory}")
  endforeach()
