cmake_minimum_required(VERSION 3.0.2)
project(autoware_behavior_velocity_stop_line_module)

find_package(catkin REQUIRED COMPONENTS geometry_msgs pluginlib roscpp visualization_msgs)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/AW149Providers.cmake")
aw149_provider(BVPC_COMMON autoware_behavior_velocity_planner_common autoware_behavior_velocity_planner_common)
aw149_provider(LANELET2_EXTENSION autoware_lanelet2_extension autoware_lanelet2_extension_lib)
aw149_provider(LANELET2_UTILS autoware_lanelet2_utils autoware_lanelet2_utils)
aw149_provider(MAP_MSGS autoware_map_msgs NONE)
aw149_provider(MOTION_UTILS autoware_motion_utils autoware_motion_utils)
aw149_provider(OBJECTS_OF_INTEREST autoware_objects_of_interest_marker_interface autoware_objects_of_interest_marker_interface)
aw149_provider(OSQP osqp osqp)
aw149_provider(OSQP_INTERFACE autoware_osqp_interface autoware_osqp_interface)
aw149_provider(PLANNING_FACTOR autoware_planning_factor_interface autoware_planning_factor_interface)
aw149_provider(QP_INTERFACE autoware_qp_interface autoware_qp_interface)
aw149_provider(ROUTE_HANDLER autoware_route_handler autoware_route_handler)
aw149_provider(TRAJECTORY autoware_trajectory autoware_trajectory)
aw149_provider(UTILS autoware_universe_utils NONE)
aw149_provider(UTILS_DEBUG autoware_utils_debug autoware_utils_debug)
aw149_provider(UTILS_GEOMETRY autoware_utils_geometry autoware_utils_geometry)
aw149_provider(UTILS_VISUALIZATION autoware_utils_visualization autoware_utils_visualization)
aw149_provider(VEHICLE_INFO autoware_vehicle_info_utils vehicle_info_utils)
aw149_provider(VELOCITY_SMOOTHER autoware_velocity_smoother autoware_velocity_smoother)
set(BVPC_COMMON_INCLUDE_DIR "" CACHE PATH "staged common include")
set(BVPC_COMMON_LIBRARY "" CACHE FILEPATH "staged common library")
set(TRAJECTORY_INCLUDE_DIR "" CACHE PATH "staged trajectory include")
set(TRAJECTORY_LIBRARY "" CACHE FILEPATH "staged trajectory library")
set(ROUTE_HANDLER_INCLUDE_DIR "" CACHE PATH "staged route-handler include")
set(ROUTE_HANDLER_LIBRARY "" CACHE FILEPATH "staged route-handler library")
set(PLANNING_FACTOR_INCLUDE_DIR "" CACHE PATH "staged planning-factor include")
set(PLANNING_FACTOR_LIBRARY "" CACHE FILEPATH "staged planning-factor library")
set(UTILS_INCLUDE_DIR "" CACHE PATH "staged utility facade include")
set(MOTION_UTILS_INCLUDE_DIR "" CACHE PATH "staged motion-utils include")
set(VEHICLE_INFO_INCLUDE_DIR "" CACHE PATH "staged vehicle-info include")
set(OBJECTS_OF_INTEREST_INCLUDE_DIR "" CACHE PATH "staged objects-of-interest include")
set(MAP_MSGS_INCLUDE_DIR "" CACHE PATH "staged ROS1 autoware-map-msgs include")
set(VELOCITY_SMOOTHER_LIBRARY "" CACHE FILEPATH "staged velocity-smoother library")
set(MOTION_UTILS_LIBRARY "" CACHE FILEPATH "staged motion-utils library")
set(UTILS_DEBUG_LIBRARY "" CACHE FILEPATH "staged utility debug library")
set(UTILS_GEOMETRY_LIBRARY "" CACHE FILEPATH "staged utility geometry library")
set(UTILS_VISUALIZATION_LIBRARY "" CACHE FILEPATH "staged utility visualization library")
set(LANELET2_UTILS_LIBRARY "" CACHE FILEPATH "staged lanelet2-utils library")
set(LANELET2_EXTENSION_LIBRARY "" CACHE FILEPATH "staged lanelet2-extension library")
set(VEHICLE_INFO_LIBRARY "" CACHE FILEPATH "staged vehicle-info library")
set(QP_INTERFACE_LIBRARY "" CACHE FILEPATH "staged qp-interface library")
set(OSQP_INTERFACE_LIBRARY "" CACHE FILEPATH "staged osqp-interface library")
set(OSQP_LIBRARY "" CACHE FILEPATH "staged osqp provider library")
foreach(path IN ITEMS BVPC_COMMON_INCLUDE_DIR BVPC_COMMON_LIBRARY TRAJECTORY_INCLUDE_DIR TRAJECTORY_LIBRARY ROUTE_HANDLER_INCLUDE_DIR ROUTE_HANDLER_LIBRARY PLANNING_FACTOR_INCLUDE_DIR PLANNING_FACTOR_LIBRARY UTILS_INCLUDE_DIR MOTION_UTILS_INCLUDE_DIR VEHICLE_INFO_INCLUDE_DIR OBJECTS_OF_INTEREST_INCLUDE_DIR MAP_MSGS_INCLUDE_DIR)
  if(NOT EXISTS "${${path}}")
    message(FATAL_ERROR "Set ${path} to its staged ROS1 provider")
  endif()
endforeach()
set(STOP_LINE_RUNTIME_LIBRARIES
  "${VELOCITY_SMOOTHER_LIBRARY}" "${MOTION_UTILS_LIBRARY}"
  "${UTILS_DEBUG_LIBRARY}" "${UTILS_GEOMETRY_LIBRARY}" "${UTILS_VISUALIZATION_LIBRARY}"
  "${LANELET2_UTILS_LIBRARY}" "${LANELET2_EXTENSION_LIBRARY}" "${VEHICLE_INFO_LIBRARY}"
  "${QP_INTERFACE_LIBRARY}" "${OSQP_INTERFACE_LIBRARY}" "${OSQP_LIBRARY}")
foreach(path IN LISTS STOP_LINE_RUNTIME_LIBRARIES)
  if(NOT EXISTS "${path}")
    message(FATAL_ERROR "Set the staged ROS1 runtime closure library: ${path}")
  endif()
endforeach()
catkin_package(LIBRARIES ${PROJECT_NAME} CATKIN_DEPENDS geometry_msgs pluginlib roscpp visualization_msgs)
# Keep ROS1-generated message headers before provider facades that also ship
# historical ROS2-style generated headers.
include_directories(src ${BVPC_COMMON_INCLUDE_DIR} ${catkin_INCLUDE_DIRS} ${MAP_MSGS_INCLUDE_DIR} ${TRAJECTORY_INCLUDE_DIR} ${ROUTE_HANDLER_INCLUDE_DIR} ${PLANNING_FACTOR_INCLUDE_DIR} ${UTILS_INCLUDE_DIR} ${MOTION_UTILS_INCLUDE_DIR} ${VEHICLE_INFO_INCLUDE_DIR} ${OBJECTS_OF_INTEREST_INCLUDE_DIR} ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
add_library(${PROJECT_NAME} SHARED
  src/debug.cpp
  src/manager.cpp
  src/scene.cpp
  src/experimental_adapter.cpp)
target_link_libraries(${PROJECT_NAME}
  "${BVPC_COMMON_LIBRARY}" "${TRAJECTORY_LIBRARY}" "${ROUTE_HANDLER_LIBRARY}"
  "${PLANNING_FACTOR_LIBRARY}" ${STOP_LINE_RUNTIME_LIBRARIES} ${catkin_LIBRARIES})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES package.xml plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})

  catkin_add_gtest(test_stop_line_scene test/test_stop_line_scene.cpp)
  if(TARGET test_stop_line_scene)
