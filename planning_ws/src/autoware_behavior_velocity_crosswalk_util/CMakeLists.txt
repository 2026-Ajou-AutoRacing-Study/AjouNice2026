cmake_minimum_required(VERSION 3.0.2)
project(autoware_behavior_velocity_crosswalk_util)
find_package(catkin REQUIRED COMPONENTS geometry_msgs roscpp)
find_package(lanelet2_core REQUIRED)
find_package(PCL REQUIRED)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/AW149Providers.cmake")
aw149_provider(BVPC_COMMON autoware_behavior_velocity_planner_common autoware_behavior_velocity_planner_common)
aw149_provider(LANELET_EXTENSION autoware_lanelet2_extension autoware_lanelet2_extension_lib)
aw149_provider(LANELET_UTILS autoware_lanelet2_utils autoware_lanelet2_utils)
aw149_provider(MAP_MSGS autoware_map_msgs NONE)
aw149_provider(MOTION_UTILS autoware_motion_utils autoware_motion_utils)
aw149_provider(OSQP osqp osqp)
aw149_provider(OSQP_INTERFACE autoware_osqp_interface autoware_osqp_interface)
aw149_provider(QP_INTERFACE autoware_qp_interface autoware_qp_interface)
aw149_provider(ROUTE_HANDLER autoware_route_handler autoware_route_handler)
aw149_provider(TRAJECTORY autoware_trajectory autoware_trajectory)
aw149_provider(UTILS autoware_universe_utils NONE)
aw149_provider(UTILS_GEOMETRY autoware_utils_geometry autoware_utils_geometry)
aw149_provider(VEHICLE_INFO autoware_vehicle_info_utils vehicle_info_utils)
aw149_provider(VELOCITY_SMOOTHER autoware_velocity_smoother autoware_velocity_smoother)
set(BVPC_COMMON_INCLUDE_DIR "" CACHE PATH "staged common include")
set(BVPC_COMMON_LIBRARY "" CACHE FILEPATH "staged common library")
set(TRAJECTORY_INCLUDE_DIR "" CACHE PATH "staged trajectory include")
set(TRAJECTORY_LIBRARY "" CACHE FILEPATH "staged trajectory library")
set(MOTION_UTILS_INCLUDE_DIR "" CACHE PATH "staged motion-utils include")
set(MOTION_UTILS_LIBRARY "" CACHE FILEPATH "staged motion-utils library")
set(LANELET_EXTENSION_INCLUDE_DIR "" CACHE PATH "staged lanelet extension include")
set(LANELET_EXTENSION_LIBRARY "" CACHE FILEPATH "staged lanelet extension library")
set(UTILS_INCLUDE_DIR "" CACHE PATH "staged universe-utils include")
set(UTILS_GEOMETRY_LIBRARY "" CACHE FILEPATH "staged universe-utils geometry library")
set(VEHICLE_INFO_INCLUDE_DIR "" CACHE PATH "staged vehicle-info include")
set(VEHICLE_INFO_LIBRARY "" CACHE FILEPATH "staged vehicle-info library")
set(ROUTE_HANDLER_LIBRARY "" CACHE FILEPATH "staged route-handler library")
set(LANELET_UTILS_LIBRARY "" CACHE FILEPATH "staged lanelet-utils library")
set(VELOCITY_SMOOTHER_LIBRARY "" CACHE FILEPATH "staged velocity-smoother library")
set(OSQP_INTERFACE_LIBRARY "" CACHE FILEPATH "staged OSQP interface library")
set(QP_INTERFACE_LIBRARY "" CACHE FILEPATH "staged QP interface library")
set(OSQP_LIBRARY "" CACHE FILEPATH "staged OSQP library")
set(MAP_MSGS_INCLUDE_DIR "" CACHE PATH "staged map messages include")
foreach(path IN ITEMS BVPC_COMMON_INCLUDE_DIR BVPC_COMMON_LIBRARY TRAJECTORY_INCLUDE_DIR TRAJECTORY_LIBRARY MOTION_UTILS_INCLUDE_DIR MOTION_UTILS_LIBRARY LANELET_EXTENSION_INCLUDE_DIR LANELET_EXTENSION_LIBRARY UTILS_INCLUDE_DIR UTILS_GEOMETRY_LIBRARY VEHICLE_INFO_INCLUDE_DIR VEHICLE_INFO_LIBRARY ROUTE_HANDLER_LIBRARY LANELET_UTILS_LIBRARY VELOCITY_SMOOTHER_LIBRARY OSQP_INTERFACE_LIBRARY QP_INTERFACE_LIBRARY OSQP_LIBRARY MAP_MSGS_INCLUDE_DIR)
  if(NOT EXISTS "${${path}}")
    message(FATAL_ERROR "Set ${path} to its staged ROS1 provider")
  endif()
endforeach()
catkin_package(INCLUDE_DIRS include LIBRARIES ${PROJECT_NAME} CATKIN_DEPENDS geometry_msgs roscpp)
include_directories(include ${catkin_INCLUDE_DIRS} ${BVPC_COMMON_INCLUDE_DIR} ${TRAJECTORY_INCLUDE_DIR} ${MOTION_UTILS_INCLUDE_DIR} ${LANELET_EXTENSION_INCLUDE_DIR} ${UTILS_INCLUDE_DIR} ${VEHICLE_INFO_INCLUDE_DIR} ${MAP_MSGS_INCLUDE_DIR} ${PCL_INCLUDE_DIRS})
add_library(${PROJECT_NAME} SHARED src/util.cpp)
target_link_libraries(${PROJECT_NAME} "${BVPC_COMMON_LIBRARY}" "${TRAJECTORY_LIBRARY}" "${ROUTE_HANDLER_LIBRARY}" "${LANELET_UTILS_LIBRARY}" "${VELOCITY_SMOOTHER_LIBRARY}" "${OSQP_INTERFACE_LIBRARY}" "${QP_INTERFACE_LIBRARY}" "${OSQP_LIBRARY}" "${MOTION_UTILS_LIBRARY}" "${LANELET_EXTENSION_LIBRARY}" "${UTILS_GEOMETRY_LIBRARY}" "${VEHICLE_INFO_LIBRARY}" lanelet2_core::lanelet2_core ${catkin_LIBRARIES})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION include)
install(FILES package.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_crosswalk_util test/test_crosswalk_util.cpp)
  if(TARGET test_crosswalk_util)
    target_link_libraries(test_crosswalk_util ${PROJECT_NAME} ${catkin_LIBRARIES})
  endif()
endif()
