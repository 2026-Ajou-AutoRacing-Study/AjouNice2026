cmake_minimum_required(VERSION 3.0.2)
project(tier4_control_mode_rviz_plugin)
find_package(catkin REQUIRED COMPONENTS autoware_vehicle_msgs pluginlib roscpp rviz)
find_package(Qt5 REQUIRED COMPONENTS Widgets)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)
catkin_package()
add_library(${PROJECT_NAME} SHARED src/control_mode_display.cpp include/control_mode_display.hpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PRIVATE include ${catkin_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME} ${catkin_LIBRARIES} Qt5::Widgets)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY plugins DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
