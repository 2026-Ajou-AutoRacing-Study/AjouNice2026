aw149_provider(UNIVERSE_UTILS autoware_universe_utils NONE)

aw149_provider(QP_INTERFACE autoware_qp_interface autoware_qp_interface)
aw149_provider(OSQP_INTERFACE autoware_osqp_interface autoware_osqp_interface)
aw149_provider(OSQP osqp osqp)
aw149_provider(UTILS_DEBUG autoware_utils_debug autoware_utils_debug)

set(LANELET2_EXTENSION_INCLUDE_DIR "" CACHE PATH "installed lanelet2-extension include directory")

set(UTILS_DEBUG_LIBRARY "" CACHE FILEPATH "installed universe-utils debug library")
foreach(path IN ITEMS MOTION_UTILS_INCLUDE_DIR MOTION_UTILS_LIBRARY OBJECT_RECOGNITION_UTILS_INCLUDE_DIR OBJECT_RECOGNITION_UTILS_LIBRARY VEHICLE_INFO_INCLUDE_DIR GEOMETRY_LIBRARY VELOCITY_SMOOTHER_INCLUDE_DIR VELOCITY_SMOOTHER_LIBRARY AUTOWARE_UTILS_RCLCPP_INCLUDE_DIR UNIVERSE_UTILS_INCLUDE_DIR ROUTE_HANDLER_INCLUDE_DIR ROUTE_HANDLER_LIBRARY LANELET2_EXTENSION_INCLUDE_DIR LANELET2_EXTENSION_LIBRARY TRAJECTORY_LIBRARY LANELET2_UTILS_LIBRARY QP_INTERFACE_LIBRARY OSQP_INTERFACE_LIBRARY OSQP_LIBRARY UTILS_DEBUG_LIBRARY)
  if(NOT EXISTS "${${path}}")
    message(FATAL_ERROR "Set ${path} to its staged ROS1 provider")
  endif()
endforeach()
catkin_package(
  INCLUDE_DIRS include
  LIBRARIES ${PROJECT_NAME}
  CATKIN_DEPENDS autoware_internal_planning_msgs autoware_perception_msgs autoware_planning_msgs
                 autoware_vehicle_info_utils geometry_msgs nav_msgs roscpp std_msgs tf2
)
include_directories(include ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${MOTION_UTILS_INCLUDE_DIR} ${OBJECT_RECOGNITION_UTILS_INCLUDE_DIR} ${VEHICLE_INFO_INCLUDE_DIR} ${VELOCITY_SMOOTHER_INCLUDE_DIR} ${AUTOWARE_UTILS_RCLCPP_INCLUDE_DIR} ${UNIVERSE_UTILS_INCLUDE_DIR} ${ROUTE_HANDLER_INCLUDE_DIR} ${LANELET2_EXTENSION_INCLUDE_DIR})
add_library(${PROJECT_NAME} SHARED
  src/original_data_constructors.cpp src/original_obstacle_helpers.cpp src/utils.cpp src/collision_checker.cpp src/polygon_utils.cpp src/planner_data.cpp)
target_link_libraries(${PROJECT_NAME} ${MOTION_UTILS_LIBRARY} ${OBJECT_RECOGNITION_UTILS_LIBRARY} ${GEOMETRY_LIBRARY} ${VELOCITY_SMOOTHER_LIBRARY} ${ROUTE_HANDLER_LIBRARY} ${LANELET2_EXTENSION_LIBRARY} ${TRAJECTORY_LIBRARY} ${LANELET2_UTILS_LIBRARY} ${QP_INTERFACE_LIBRARY} ${OSQP_INTERFACE_LIBRARY} ${OSQP_LIBRARY} ${UTILS_DEBUG_LIBRARY} ${catkin_LIBRARIES} ${PCL_LIBRARIES})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
