cmake_minimum_required(VERSION 3.0.2)
project(autoware_motion_velocity_dynamic_obstacle_stop_module)
find_package(catkin REQUIRED COMPONENTS autoware_perception_msgs autoware_planning_msgs geometry_msgs pluginlib roscpp tf2 visualization_msgs)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/AW149Providers.cmake")
aw149_provider(AUTOWARE_UTILS_RCLCPP autoware_utils_rclcpp NONE)
aw149_provider(MOTION_UTILS autoware_motion_utils autoware_motion_utils)
aw149_provider(MVP_COMMON autoware_motion_velocity_planner_common autoware_motion_velocity_planner_common)
aw149_provider(OBJECT_RECOGNITION_UTILS autoware_object_recognition_utils autoware_object_recognition_utils)
aw149_provider(PLANNING_FACTOR autoware_planning_factor_interface autoware_planning_factor_interface)
aw149_provider(UNIVERSE_UTILS autoware_universe_utils NONE)
aw149_provider(UNIVERSE_UTILS_DEBUG autoware_utils_debug autoware_utils_debug)
aw149_provider(VEHICLE_INFO autoware_vehicle_info_utils vehicle_info_utils)
aw149_provider(VELOCITY_SMOOTHER autoware_velocity_smoother autoware_velocity_smoother)
set(MVP_COMMON_INCLUDE_DIR "" CACHE PATH "installed MVP common include directory")
set(MVP_COMMON_LIBRARY "" CACHE FILEPATH "installed MVP common library")
set(MOTION_UTILS_INCLUDE_DIR "" CACHE PATH "installed motion-utils include directory")
set(MOTION_UTILS_LIBRARY "" CACHE FILEPATH "installed motion-utils library")
set(OBJECT_RECOGNITION_UTILS_INCLUDE_DIR "" CACHE PATH "installed object-recognition include directory")
set(OBJECT_RECOGNITION_UTILS_LIBRARY "" CACHE FILEPATH "installed object-recognition library")
set(VEHICLE_INFO_INCLUDE_DIR "" CACHE PATH "installed vehicle-info include directory")
set(VEHICLE_INFO_LIBRARY "" CACHE FILEPATH "installed vehicle-info library")
set(UNIVERSE_UTILS_INCLUDE_DIR "" CACHE PATH "installed universe-utils include directory")
set(UNIVERSE_UTILS_DEBUG_LIBRARY "" CACHE FILEPATH "installed universe-utils debug library")
set(PLANNING_FACTOR_INCLUDE_DIR "" CACHE PATH "installed planning-factor include directory")
set(PLANNING_FACTOR_LIBRARY "" CACHE FILEPATH "installed planning-factor library")
set(VELOCITY_SMOOTHER_INCLUDE_DIR "" CACHE PATH "installed velocity-smoother include directory")
set(VELOCITY_SMOOTHER_LIBRARY "" CACHE FILEPATH "installed velocity-smoother library")
set(AUTOWARE_UTILS_RCLCPP_INCLUDE_DIR "" CACHE PATH "installed ROS1 parameter facade include directory")
foreach(path IN ITEMS MVP_COMMON_INCLUDE_DIR MVP_COMMON_LIBRARY MOTION_UTILS_INCLUDE_DIR MOTION_UTILS_LIBRARY OBJECT_RECOGNITION_UTILS_INCLUDE_DIR OBJECT_RECOGNITION_UTILS_LIBRARY VEHICLE_INFO_INCLUDE_DIR VEHICLE_INFO_LIBRARY UNIVERSE_UTILS_INCLUDE_DIR UNIVERSE_UTILS_DEBUG_LIBRARY PLANNING_FACTOR_INCLUDE_DIR PLANNING_FACTOR_LIBRARY VELOCITY_SMOOTHER_INCLUDE_DIR VELOCITY_SMOOTHER_LIBRARY AUTOWARE_UTILS_RCLCPP_INCLUDE_DIR)
  if(NOT EXISTS "${${path}}")
    message(FATAL_ERROR "Set ${path} to its staged ROS1 provider")
  endif()
endforeach()
catkin_package(LIBRARIES ${PROJECT_NAME} CATKIN_DEPENDS autoware_perception_msgs autoware_planning_msgs geometry_msgs pluginlib roscpp tf2 visualization_msgs)
include_directories(src ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${MVP_COMMON_INCLUDE_DIR} ${MOTION_UTILS_INCLUDE_DIR} ${OBJECT_RECOGNITION_UTILS_INCLUDE_DIR} ${VEHICLE_INFO_INCLUDE_DIR} ${UNIVERSE_UTILS_INCLUDE_DIR} ${PLANNING_FACTOR_INCLUDE_DIR} ${VELOCITY_SMOOTHER_INCLUDE_DIR} ${AUTOWARE_UTILS_RCLCPP_INCLUDE_DIR})
file(GLOB MODULE_SOURCES src/*.cpp)
add_library(${PROJECT_NAME} SHARED ${MODULE_SOURCES})
target_link_libraries(${PROJECT_NAME} ${MVP_COMMON_LIBRARY} ${MOTION_UTILS_LIBRARY} ${OBJECT_RECOGNITION_UTILS_LIBRARY} ${VEHICLE_INFO_LIBRARY} ${UNIVERSE_UTILS_DEBUG_LIBRARY} ${PLANNING_FACTOR_LIBRARY} ${VELOCITY_SMOOTHER_LIBRARY} ${catkin_LIBRARIES})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
add_executable(verify_dynamic_obstacle_stop_plugin test/pluginlib_r023.cpp)
target_link_libraries(verify_dynamic_obstacle_stop_plugin ${catkin_LIBRARIES})
target_compile_features(verify_dynamic_obstacle_stop_plugin PRIVATE cxx_std_17)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES package.xml plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY config/ DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/config)
