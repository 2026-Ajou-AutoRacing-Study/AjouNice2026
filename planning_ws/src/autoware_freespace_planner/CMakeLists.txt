cmake_minimum_required(VERSION 3.13)
project(autoware_freespace_planner)
find_package(catkin REQUIRED COMPONENTS roscpp std_msgs geometry_msgs nav_msgs tf2_ros tf2_geometry_msgs autoware_planning_msgs autoware_internal_planning_msgs autoware_internal_debug_msgs autoware_freespace_planning_algorithms autoware_vehicle_info_utils autoware_motion_utils autoware_utils)
catkin_package(INCLUDE_DIRS include LIBRARIES ${PROJECT_NAME}_utils CATKIN_DEPENDS roscpp std_msgs geometry_msgs nav_msgs tf2_ros tf2_geometry_msgs autoware_planning_msgs autoware_internal_planning_msgs autoware_internal_debug_msgs autoware_freespace_planning_algorithms autoware_vehicle_info_utils autoware_motion_utils autoware_utils)
include_directories(include ${catkin_INCLUDE_DIRS})
add_library(${PROJECT_NAME}_utils src/utils.cpp src/freespace_planner_node.cpp)
target_compile_features(${PROJECT_NAME}_utils PUBLIC cxx_std_17)
target_link_libraries(${PROJECT_NAME}_utils ${catkin_LIBRARIES})
target_link_options(${PROJECT_NAME}_utils PRIVATE -Wl,--no-undefined)
add_executable(freespace_planner_node src/main.cpp)
target_link_libraries(freespace_planner_node ${PROJECT_NAME}_utils ${catkin_LIBRARIES})
target_compile_features(freespace_planner_node PRIVATE cxx_std_17)
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_freespace_planner_ros1_node test/test_freespace_planner_ros1_node.cpp test/test_freespace_replanning.cpp)
  if(TARGET test_freespace_planner_ros1_node)
    target_link_libraries(test_freespace_planner_ros1_node ${PROJECT_NAME}_utils ${catkin_LIBRARIES})
  endif()
endif()
install(TARGETS ${PROJECT_NAME}_utils freespace_planner_node ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
install(DIRECTORY config launch DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
