cmake_minimum_required(VERSION 3.0.2)
project(autoware_motion_velocity_road_user_stop_module)

find_package(catkin REQUIRED COMPONENTS autoware_internal_debug_msgs autoware_internal_planning_msgs autoware_perception_msgs autoware_planning_msgs geometry_msgs nav_msgs pluginlib roscpp std_msgs tf2 tf2_geometry_msgs unique_identifier_msgs visualization_msgs)
find_package(lanelet2_core REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/AW149Providers.cmake")
aw149_provider(AUTOWARE_UTILS_RCLCPP autoware_utils_rclcpp NONE)
aw149_provider(GEOMETRY autoware_utils_geometry autoware_utils_geometry)
aw149_provider(LANELET2_UTILS autoware_lanelet2_utils autoware_lanelet2_utils)
aw149_provider(MOTION_UTILS autoware_motion_utils autoware_motion_utils)
aw149_provider(MVP_COMMON autoware_motion_velocity_planner_common autoware_motion_velocity_planner_common)
aw149_provider(OBJECT_RECOGNITION_UTILS autoware_object_recognition_utils autoware_object_recognition_utils)
aw149_provider(PLANNING_FACTOR_INTERFACE autoware_planning_factor_interface autoware_planning_factor_interface)
aw149_provider(ROUTE_HANDLER autoware_route_handler autoware_route_handler)
aw149_provider(UNIVERSE_UTILS autoware_universe_utils NONE)
aw149_provider(UNIVERSE_UTILS_DEBUG autoware_utils_debug autoware_utils_debug)
aw149_provider(VEHICLE_INFO autoware_vehicle_info_utils vehicle_info_utils)
set(MVP_COMMON_INCLUDE_DIR "" CACHE PATH "installed MVP common include directory")
set(MVP_COMMON_LIBRARY "" CACHE FILEPATH "installed MVP common library")
set(MOTION_UTILS_INCLUDE_DIR "" CACHE PATH "installed motion-utils include directory")
set(MOTION_UTILS_LIBRARY "" CACHE FILEPATH "installed motion-utils library")
set(OBJECT_RECOGNITION_UTILS_LIBRARY "" CACHE FILEPATH "installed object-recognition-utils library")
set(VEHICLE_INFO_INCLUDE_DIR "" CACHE PATH "installed vehicle-info include directory")
set(GEOMETRY_INCLUDE_DIR "" CACHE PATH "installed geometry include directory")
set(GEOMETRY_LIBRARY "" CACHE FILEPATH "installed geometry library")
set(ROUTE_HANDLER_INCLUDE_DIR "" CACHE PATH "installed route-handler include directory")
set(ROUTE_HANDLER_LIBRARY "" CACHE FILEPATH "installed route-handler library")
set(AUTOWARE_UTILS_RCLCPP_INCLUDE_DIR "" CACHE PATH "installed ROS1 parameter facade include directory")
set(UNIVERSE_UTILS_INCLUDE_DIR "" CACHE PATH "installed ROS1 universe-utils include directory")
set(UNIVERSE_UTILS_DEBUG_LIBRARY "" CACHE FILEPATH "installed ROS1 universe-utils debug library")
set(PLANNING_FACTOR_INTERFACE_INCLUDE_DIR "" CACHE PATH "installed planning-factor interface include directory")
set(PLANNING_FACTOR_INTERFACE_LIBRARY "" CACHE FILEPATH "installed planning-factor interface library")
set(LANELET2_UTILS_INCLUDE_DIR "" CACHE PATH "installed lanelet2-utils include directory")
set(LANELET2_UTILS_LIBRARY "" CACHE FILEPATH "installed lanelet2-utils library")
foreach(path IN ITEMS MVP_COMMON_INCLUDE_DIR MVP_COMMON_LIBRARY MOTION_UTILS_INCLUDE_DIR MOTION_UTILS_LIBRARY OBJECT_RECOGNITION_UTILS_LIBRARY VEHICLE_INFO_INCLUDE_DIR GEOMETRY_INCLUDE_DIR GEOMETRY_LIBRARY ROUTE_HANDLER_INCLUDE_DIR ROUTE_HANDLER_LIBRARY AUTOWARE_UTILS_RCLCPP_INCLUDE_DIR UNIVERSE_UTILS_INCLUDE_DIR UNIVERSE_UTILS_DEBUG_LIBRARY PLANNING_FACTOR_INTERFACE_INCLUDE_DIR PLANNING_FACTOR_INTERFACE_LIBRARY LANELET2_UTILS_INCLUDE_DIR LANELET2_UTILS_LIBRARY)
  if(NOT EXISTS "${${path}}")
    message(FATAL_ERROR "Set ${path} to an explicit staged ROS1 provider")
  endif()
endforeach()
catkin_package(INCLUDE_DIRS src LIBRARIES ${PROJECT_NAME} CATKIN_DEPENDS autoware_internal_debug_msgs autoware_internal_planning_msgs autoware_perception_msgs autoware_planning_msgs geometry_msgs nav_msgs pluginlib roscpp std_msgs tf2 tf2_geometry_msgs unique_identifier_msgs visualization_msgs)
include_directories(src ${catkin_INCLUDE_DIRS} ${lanelet2_core_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${MVP_COMMON_INCLUDE_DIR} ${MOTION_UTILS_INCLUDE_DIR} ${VEHICLE_INFO_INCLUDE_DIR} ${GEOMETRY_INCLUDE_DIR} ${ROUTE_HANDLER_INCLUDE_DIR} ${AUTOWARE_UTILS_RCLCPP_INCLUDE_DIR} ${UNIVERSE_UTILS_INCLUDE_DIR} ${PLANNING_FACTOR_INTERFACE_INCLUDE_DIR} ${LANELET2_UTILS_INCLUDE_DIR})
add_library(${PROJECT_NAME} SHARED src/debug.cpp src/road_user_stop_module.cpp src/utils.cpp)
target_link_libraries(${PROJECT_NAME} ${MVP_COMMON_LIBRARY} ${MOTION_UTILS_LIBRARY} ${OBJECT_RECOGNITION_UTILS_LIBRARY} ${GEOMETRY_LIBRARY} ${ROUTE_HANDLER_LIBRARY} ${UNIVERSE_UTILS_DEBUG_LIBRARY} ${PLANNING_FACTOR_INTERFACE_LIBRARY} ${LANELET2_UTILS_LIBRARY} ${catkin_LIBRARIES} ${lanelet2_core_LIBRARIES} ${PCL_LIBRARIES})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_plugin_factory test/test_plugin_factory.cpp)
  if(TARGET test_plugin_factory)
    target_link_libraries(test_plugin_factory ${PROJECT_NAME} ${catkin_LIBRARIES})
    target_compile_definitions(test_plugin_factory PRIVATE ROAD_USER_STOP_LIBRARY_PATH="$<TARGET_FILE:${PROJECT_NAME}>")
  endif()
endif()
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES package.xml plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
