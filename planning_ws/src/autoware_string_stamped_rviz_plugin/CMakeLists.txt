cmake_minimum_required(VERSION 3.0.2)
project(autoware_string_stamped_rviz_plugin)
find_package(catkin REQUIRED COMPONENTS autoware_internal_debug_msgs pluginlib roscpp rviz rviz_2d_overlay_plugins rviz_2d_overlay_msgs)
find_package(Qt5 REQUIRED COMPONENTS Core Gui Widgets)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)
add_definitions(-DQT_NO_KEYWORDS)
catkin_package()
qt5_wrap_cpp(MOC_FILES src/string_stamped_overlay_display.hpp)
add_library(${PROJECT_NAME} SHARED ${MOC_FILES} src/string_stamped_overlay_display.cpp src/string_stamped_overlay_display.hpp)
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PRIVATE src ${catkin_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME} ${catkin_LIBRARIES} Qt5::Core Qt5::Gui Qt5::Widgets)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES plugins/plugins_description.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
