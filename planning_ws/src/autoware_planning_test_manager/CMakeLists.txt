cmake_minimum_required(VERSION 3.0.2)
project(autoware_planning_test_manager)
set(CMAKE_CXX_STANDARD 17)
find_package(catkin REQUIRED COMPONENTS autoware_motion_utils autoware_planning_msgs autoware_internal_planning_msgs autoware_test_utils nav_msgs roscpp)
catkin_package(INCLUDE_DIRS include LIBRARIES autoware_planning_test_manager CATKIN_DEPENDS autoware_motion_utils autoware_planning_msgs autoware_internal_planning_msgs autoware_test_utils nav_msgs roscpp)
add_library(autoware_planning_test_manager src/autoware_planning_test_manager.cpp)
target_include_directories(autoware_planning_test_manager PUBLIC include ${catkin_INCLUDE_DIRS})
target_link_libraries(autoware_planning_test_manager PUBLIC ${catkin_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_autoware_planning_test_manager test/test_planning_test_manager.cpp)
  target_link_libraries(test_autoware_planning_test_manager autoware_planning_test_manager ${catkin_LIBRARIES})
endif()
install(TARGETS autoware_planning_test_manager ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/autoware/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION}/autoware)
