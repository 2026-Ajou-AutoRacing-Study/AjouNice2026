cmake_minimum_required(VERSION 3.0.2)
project(autoware_bezier_sampler)
find_package(catkin REQUIRED COMPONENTS autoware_sampler_common)
find_package(Eigen3 REQUIRED)
catkin_package(INCLUDE_DIRS include LIBRARIES ${PROJECT_NAME} CATKIN_DEPENDS autoware_sampler_common DEPENDS EIGEN3)
add_library(${PROJECT_NAME} SHARED src/bezier.cpp src/bezier_sampling.cpp)
target_include_directories(${PROJECT_NAME} PUBLIC $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}/include> $<INSTALL_INTERFACE:${CATKIN_GLOBAL_INCLUDE_DESTINATION}> ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_link_libraries(${PROJECT_NAME} PUBLIC ${catkin_LIBRARIES} Eigen3::Eigen)
if(CATKIN_ENABLE_TESTING)
  # ROS2 BUILD_TESTING is mapped to an executable ROS1 contract test.  It
  # protects the observed upstream vector-constructor storage/layout result.
  catkin_add_gtest(test_vector_layout test/test_vector_layout.cpp)
  if(TARGET test_vector_layout)
    target_compile_features(test_vector_layout PRIVATE cxx_std_17)
    target_link_libraries(test_vector_layout ${PROJECT_NAME} ${catkin_LIBRARIES} Eigen3::Eigen gtest_main)
  endif()
endif()
install(TARGETS ${PROJECT_NAME} ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
