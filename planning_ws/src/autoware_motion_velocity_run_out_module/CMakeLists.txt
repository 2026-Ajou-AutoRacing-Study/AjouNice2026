cmake_minimum_required(VERSION 3.0.2)
project(autoware_motion_velocity_run_out_module)

find_package(catkin REQUIRED COMPONENTS autoware_internal_debug_msgs autoware_internal_planning_msgs autoware_lanelet2_utils autoware_map_msgs autoware_motion_velocity_planner autoware_perception_msgs autoware_planning_msgs diagnostic_updater geometry_msgs nav_msgs pluginlib roscpp std_msgs tf2 tf2_geometry_msgs visualization_msgs)
find_package(lanelet2_core REQUIRED)
find_package(PCL REQUIRED COMPONENTS common)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/AW149Providers.cmake")
aw149_provider(MOTION_UTILS autoware_motion_utils autoware_motion_utils)
aw149_provider(MVP_COMMON autoware_motion_velocity_planner_common autoware_motion_velocity_planner_common)
aw149_provider(OBJECT_MARKER autoware_objects_of_interest_marker_interface autoware_objects_of_interest_marker_interface)
aw149_provider(PLANNING_FACTOR_INTERFACE autoware_planning_factor_interface autoware_planning_factor_interface)
aw149_provider(ROUTE_HANDLER autoware_route_handler autoware_route_handler)
aw149_provider(UNIVERSE_UTILS autoware_universe_utils autoware_universe_utils)
aw149_provider(UNIVERSE_UTILS_DEBUG autoware_utils_debug autoware_utils_debug)
aw149_provider(VEHICLE_INFO autoware_vehicle_info_utils vehicle_info_utils)
set(MVP_COMMON_INCLUDE_DIR "" CACHE PATH "installed MVP common include directory")
set(MVP_COMMON_LIBRARY "" CACHE FILEPATH "installed MVP common library")
set(MOTION_UTILS_INCLUDE_DIR "" CACHE PATH "installed motion-utils include directory")
set(MOTION_UTILS_LIBRARY "" CACHE FILEPATH "installed motion-utils library")
set(ROUTE_HANDLER_INCLUDE_DIR "" CACHE PATH "installed route-handler include directory")
set(ROUTE_HANDLER_LIBRARY "" CACHE FILEPATH "installed route-handler library")
set(UNIVERSE_UTILS_INCLUDE_DIR "" CACHE PATH "installed universe-utils facade include directory")
set(UNIVERSE_UTILS_LIBRARY "" CACHE FILEPATH "installed universe-utils facade library")
set(UNIVERSE_UTILS_DEBUG_LIBRARY "" CACHE FILEPATH "installed debug provider library")
set(OBJECT_MARKER_INCLUDE_DIR "" CACHE PATH "installed object-marker include directory")
set(OBJECT_MARKER_LIBRARY "" CACHE FILEPATH "installed object-marker library")
set(PLANNING_FACTOR_INTERFACE_INCLUDE_DIR "" CACHE PATH "installed planning-factor include directory")
set(PLANNING_FACTOR_INTERFACE_LIBRARY "" CACHE FILEPATH "installed planning-factor library")
set(VEHICLE_INFO_INCLUDE_DIR "" CACHE PATH "installed vehicle-info include directory")
set(VEHICLE_INFO_LIBRARY "" CACHE FILEPATH "installed vehicle-info library")
foreach(path IN ITEMS MVP_COMMON_INCLUDE_DIR MVP_COMMON_LIBRARY MOTION_UTILS_INCLUDE_DIR MOTION_UTILS_LIBRARY ROUTE_HANDLER_INCLUDE_DIR ROUTE_HANDLER_LIBRARY UNIVERSE_UTILS_INCLUDE_DIR UNIVERSE_UTILS_LIBRARY UNIVERSE_UTILS_DEBUG_LIBRARY OBJECT_MARKER_INCLUDE_DIR OBJECT_MARKER_LIBRARY PLANNING_FACTOR_INTERFACE_INCLUDE_DIR PLANNING_FACTOR_INTERFACE_LIBRARY VEHICLE_INFO_INCLUDE_DIR VEHICLE_INFO_LIBRARY)
  if(NOT EXISTS "${${path}}")
    message(FATAL_ERROR "Set ${path} to an explicit staged ROS1 provider")
  endif()
endforeach()
catkin_package(INCLUDE_DIRS src LIBRARIES ${PROJECT_NAME} CATKIN_DEPENDS autoware_internal_debug_msgs autoware_internal_planning_msgs autoware_lanelet2_utils autoware_map_msgs autoware_motion_velocity_planner autoware_perception_msgs autoware_planning_msgs diagnostic_updater geometry_msgs nav_msgs pluginlib roscpp std_msgs tf2 tf2_geometry_msgs visualization_msgs)
include_directories(src ${catkin_INCLUDE_DIRS} ${lanelet2_core_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${MVP_COMMON_INCLUDE_DIR} ${MOTION_UTILS_INCLUDE_DIR} ${ROUTE_HANDLER_INCLUDE_DIR} ${UNIVERSE_UTILS_INCLUDE_DIR} ${OBJECT_MARKER_INCLUDE_DIR} ${PLANNING_FACTOR_INTERFACE_INCLUDE_DIR} ${VEHICLE_INFO_INCLUDE_DIR})
file(GLOB RUN_OUT_SOURCES src/*.cpp)
add_library(${PROJECT_NAME} SHARED ${RUN_OUT_SOURCES})
target_link_libraries(${PROJECT_NAME} ${MVP_COMMON_LIBRARY} ${MOTION_UTILS_LIBRARY} ${ROUTE_HANDLER_LIBRARY} ${UNIVERSE_UTILS_LIBRARY} ${UNIVERSE_UTILS_DEBUG_LIBRARY} ${OBJECT_MARKER_LIBRARY} ${PLANNING_FACTOR_INTERFACE_LIBRARY} ${VEHICLE_INFO_LIBRARY} ${catkin_LIBRARIES} ${lanelet2_core_LIBRARIES} ${PCL_LIBRARIES})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_plugin_factory test/test_plugin_factory.cpp)
  if(TARGET test_plugin_factory)
    target_link_libraries(test_plugin_factory ${PROJECT_NAME} ${catkin_LIBRARIES})
    target_compile_definitions(test_plugin_factory PRIVATE RUN_OUT_LIBRARY_PATH="$<TARGET_FILE:${PROJECT_NAME}>")
  endif()
  add_executable(run_out_host_map_probe test/run_out_host_map_probe.cpp)
  target_link_libraries(run_out_host_map_probe ${catkin_LIBRARIES})
  target_compile_features(run_out_host_map_probe PRIVATE cxx_std_17)
endif()
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES package.xml plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES config/run_out.ros1.param.yaml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/config)
