cmake_minimum_required(VERSION 3.0.2)
project(tier4_dummy_object_rviz_plugin)
find_package(catkin REQUIRED COMPONENTS autoware_perception_msgs geometry_msgs pluginlib roscpp rviz tf2 tf2_geometry_msgs tf2_ros tier4_simulation_msgs)
find_package(Qt5 REQUIRED COMPONENTS Core Widgets)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)
catkin_package()
file(GLOB DUMMY_OBJECT_SOURCES src/tools/*.cpp src/tools/*.hpp)
add_library(${PROJECT_NAME} SHARED ${DUMMY_OBJECT_SOURCES})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PRIVATE src/tools ${catkin_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME} ${catkin_LIBRARIES} Qt5::Core Qt5::Widgets)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY icons plugins images DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
