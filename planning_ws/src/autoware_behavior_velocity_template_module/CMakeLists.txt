cmake_minimum_required(VERSION 3.0.2)
project(autoware_behavior_velocity_template_module)

find_package(catkin REQUIRED COMPONENTS pluginlib roscpp visualization_msgs)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/AW149Providers.cmake")
aw149_provider(BVPC_COMMON autoware_behavior_velocity_planner_common autoware_behavior_velocity_planner_common)
aw149_provider(LANELET_UTILS autoware_lanelet2_utils autoware_lanelet2_utils)
aw149_provider(MAP_MSGS autoware_map_msgs NONE)
aw149_provider(MOTION_UTILS autoware_motion_utils autoware_motion_utils)
aw149_provider(OBJECTS_OF_INTEREST autoware_objects_of_interest_marker_interface autoware_objects_of_interest_marker_interface)
aw149_provider(PLANNING_FACTOR autoware_planning_factor_interface autoware_planning_factor_interface)
aw149_provider(TRAJECTORY autoware_trajectory autoware_trajectory)
aw149_provider(UTILS autoware_universe_utils NONE)
aw149_provider(UTILS_DEBUG autoware_utils_debug autoware_utils_debug)
aw149_provider(VEHICLE_INFO autoware_vehicle_info_utils vehicle_info_utils)
set(BVPC_COMMON_INCLUDE_DIR "" CACHE PATH "staged common include")
set(BVPC_COMMON_LIBRARY "" CACHE FILEPATH "staged common library")
set(TRAJECTORY_INCLUDE_DIR "" CACHE PATH "staged trajectory include")
set(TRAJECTORY_LIBRARY "" CACHE FILEPATH "staged trajectory library")
set(PLANNING_FACTOR_INCLUDE_DIR "" CACHE PATH "staged planning-factor include")
set(PLANNING_FACTOR_LIBRARY "" CACHE FILEPATH "staged planning-factor library")
set(MAP_MSGS_INCLUDE_DIR "" CACHE PATH "ROS1 autoware map-message include")
set(VEHICLE_INFO_INCLUDE_DIR "" CACHE PATH "staged vehicle-info include")
set(MOTION_UTILS_INCLUDE_DIR "" CACHE PATH "staged motion-utils include")
set(MOTION_UTILS_LIBRARY "" CACHE FILEPATH "staged motion-utils library")
set(OBJECTS_OF_INTEREST_INCLUDE_DIR "" CACHE PATH "staged objects-of-interest include")
set(UTILS_INCLUDE_DIR "" CACHE PATH "staged utility facade include")
set(UTILS_DEBUG_LIBRARY "" CACHE FILEPATH "staged utility debug library")
set(LANELET_UTILS_LIBRARY "" CACHE FILEPATH "staged lanelet-utils library")
foreach(path IN ITEMS BVPC_COMMON_INCLUDE_DIR BVPC_COMMON_LIBRARY TRAJECTORY_INCLUDE_DIR TRAJECTORY_LIBRARY PLANNING_FACTOR_INCLUDE_DIR PLANNING_FACTOR_LIBRARY MAP_MSGS_INCLUDE_DIR VEHICLE_INFO_INCLUDE_DIR MOTION_UTILS_INCLUDE_DIR MOTION_UTILS_LIBRARY OBJECTS_OF_INTEREST_INCLUDE_DIR UTILS_INCLUDE_DIR UTILS_DEBUG_LIBRARY LANELET_UTILS_LIBRARY)
  if(NOT EXISTS "${${path}}")
    message(FATAL_ERROR "Set ${path} to its staged ROS1 provider")
  endif()
endforeach()

catkin_package(LIBRARIES ${PROJECT_NAME} CATKIN_DEPENDS pluginlib roscpp visualization_msgs)
include_directories(src ${BVPC_COMMON_INCLUDE_DIR} ${TRAJECTORY_INCLUDE_DIR} ${PLANNING_FACTOR_INCLUDE_DIR} ${MAP_MSGS_INCLUDE_DIR} ${VEHICLE_INFO_INCLUDE_DIR} ${MOTION_UTILS_INCLUDE_DIR} ${OBJECTS_OF_INTEREST_INCLUDE_DIR} ${UTILS_INCLUDE_DIR} ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${catkin_INCLUDE_DIRS})
add_library(${PROJECT_NAME} SHARED src/manager.cpp src/scene.cpp)
target_link_libraries(${PROJECT_NAME} "${BVPC_COMMON_LIBRARY}" "${TRAJECTORY_LIBRARY}" "${PLANNING_FACTOR_LIBRARY}" "${MOTION_UTILS_LIBRARY}" "${UTILS_DEBUG_LIBRARY}" "${LANELET_UTILS_LIBRARY}" ${catkin_LIBRARIES})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES package.xml plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})

if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_template_plugin test/test_template_plugin.cpp)
  if(TARGET test_template_plugin)
    target_link_libraries(test_template_plugin ${PROJECT_NAME} ${catkin_LIBRARIES})
  endif()
endif()
