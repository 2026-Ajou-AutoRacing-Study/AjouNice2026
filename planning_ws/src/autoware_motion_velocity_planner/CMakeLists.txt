  endif()
  catkin_add_gtest(test_motion_velocity_planner_manager_core test/test_motion_velocity_planner_manager_core.cpp)
  if(TARGET test_motion_velocity_planner_manager_core)
    target_link_libraries(test_motion_velocity_planner_manager_core ${PROJECT_NAME} ${catkin_LIBRARIES})
  endif()
  catkin_add_gtest(test_motion_velocity_planner_manager test/test_motion_velocity_planner_manager.cpp)
  if(TARGET test_motion_velocity_planner_manager)
    target_link_libraries(test_motion_velocity_planner_manager ${PROJECT_NAME} ${catkin_LIBRARIES})
  endif()

  # Keep the node/plugin tests hermetic to this candidate closure.  catkin's
  # cached environment still contains the older common and obstacle-stop
  # installs used at configure time; without an explicit ordering pluginlib
  # can resolve a stale manifest while the host itself links the clean ABI.
  get_filename_component(AW149_COMMON_PREFIX "${autoware_motion_velocity_planner_common_DIR}/../../.." ABSOLUTE)
  set(AW149_MVP_TEST_PREFIXES
    "${CATKIN_DEVEL_PREFIX}"
    "${AW149_COMMON_PREFIX}")
  list(REMOVE_DUPLICATES AW149_MVP_TEST_PREFIXES)
  set(AW149_MVP_TEST_ROS_PATHS "${CMAKE_CURRENT_SOURCE_DIR}")
  set(AW149_MVP_TEST_LIBRARY_PATHS "")
  foreach(prefix IN LISTS AW149_MVP_TEST_PREFIXES)
    if(NOT IS_DIRECTORY "${prefix}")
      message(FATAL_ERROR "MVP test closure prefix is absent: ${prefix}")
    endif()
    list(APPEND AW149_MVP_TEST_LIBRARY_PATHS "${prefix}/lib")
    if(IS_DIRECTORY "${prefix}/share")
      list(APPEND AW149_MVP_TEST_ROS_PATHS "${prefix}/share")
    endif()
  endforeach()
  list(APPEND AW149_MVP_TEST_ROS_PATHS "/opt/ros/noetic/share")
  list(APPEND AW149_MVP_TEST_LIBRARY_PATHS "/opt/ros/noetic/lib" "/opt/ros/noetic/lib/x86_64-linux-gnu")
  string(JOIN ":" AW149_MVP_TEST_PREFIX_PATH ${AW149_MVP_TEST_PREFIXES})
  string(JOIN ":" AW149_MVP_TEST_ROS_PATH ${AW149_MVP_TEST_ROS_PATHS})
  string(JOIN ":" AW149_MVP_TEST_LIBRARY_PATH ${AW149_MVP_TEST_LIBRARY_PATHS})
  foreach(test_name IN ITEMS
      _ctest_${PROJECT_NAME}_gtest_test_motion_velocity_planner_core
      _ctest_${PROJECT_NAME}_gtest_test_motion_velocity_planner_ros1_node
      _ctest_${PROJECT_NAME}_gtest_test_motion_velocity_planner_manager_core
      _ctest_${PROJECT_NAME}_gtest_test_motion_velocity_planner_manager)
    set_tests_properties(${test_name} PROPERTIES ENVIRONMENT
      "CMAKE_PREFIX_PATH=${AW149_MVP_TEST_PREFIX_PATH}:$ENV{CMAKE_PREFIX_PATH};ROS_PACKAGE_PATH=${AW149_MVP_TEST_ROS_PATH}:$ENV{ROS_PACKAGE_PATH};LD_LIBRARY_PATH=${AW149_MVP_TEST_LIBRARY_PATH}:$ENV{LD_LIBRARY_PATH}")
  endforeach()
endif()
