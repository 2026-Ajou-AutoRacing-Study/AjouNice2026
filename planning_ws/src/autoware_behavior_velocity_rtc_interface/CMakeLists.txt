cmake_minimum_required(VERSION 3.0.2)
project(autoware_behavior_velocity_rtc_interface)
find_package(catkin REQUIRED COMPONENTS roscpp tier4_rtc_msgs unique_identifier_msgs)
find_package(Eigen3 REQUIRED)
find_package(PCL REQUIRED)
include("${CMAKE_CURRENT_LIST_DIR}/../cmake/AW149Providers.cmake")
aw149_provider(BVPC_COMMON autoware_behavior_velocity_planner_common autoware_behavior_velocity_planner_common)
aw149_provider(MAP_MSGS autoware_map_msgs NONE)
aw149_provider(MOTION_UTILS autoware_motion_utils autoware_motion_utils)
aw149_provider(OBJECTS autoware_objects_of_interest_marker_interface autoware_objects_of_interest_marker_interface)
aw149_provider(PLANNING_FACTOR autoware_planning_factor_interface autoware_planning_factor_interface)
aw149_provider(ROUTE_HANDLER autoware_route_handler autoware_route_handler)
aw149_provider(RTC autoware_rtc_interface autoware_rtc_interface)
aw149_provider(TRAJECTORY autoware_trajectory autoware_trajectory)
aw149_provider(UTILS autoware_universe_utils NONE)
aw149_provider(UTILS_DEBUG autoware_utils_debug autoware_utils_debug)
aw149_provider(VEHICLE_INFO autoware_vehicle_info_utils vehicle_info_utils)
aw149_provider(VELOCITY_SMOOTHER autoware_velocity_smoother autoware_velocity_smoother)
set(BVPC_COMMON_INCLUDE_DIR "" CACHE PATH "staged common include")
set(BVPC_COMMON_LIBRARY "" CACHE FILEPATH "staged common library")
set(VELOCITY_SMOOTHER_LIBRARY "" CACHE FILEPATH "staged velocity smoother library")
set(VEHICLE_INFO_LIBRARY "" CACHE FILEPATH "staged vehicle-info library")
set(ROUTE_HANDLER_LIBRARY "" CACHE FILEPATH "staged route-handler library")
set(RTC_INCLUDE_DIR "" CACHE PATH "staged RTC include")
set(RTC_LIBRARY "" CACHE FILEPATH "staged RTC library")
set(OBJECTS_INCLUDE_DIR "" CACHE PATH "staged objects marker include")
set(UTILS_INCLUDE_DIR "" CACHE PATH "staged utility include")
set(UTILS_DEBUG_LIBRARY "" CACHE FILEPATH "staged utility debug library")
set(VEHICLE_INFO_INCLUDE_DIR "" CACHE PATH "staged vehicle-info include")
set(MAP_MSGS_INCLUDE_DIR "" CACHE PATH "ROS1 map-message include")
set(MOTION_UTILS_INCLUDE_DIR "" CACHE PATH "staged motion-utils include")
set(TRAJECTORY_INCLUDE_DIR "" CACHE PATH "staged trajectory include")
set(PLANNING_FACTOR_INCLUDE_DIR "" CACHE PATH "staged planning-factor include")
set(MOTION_UTILS_LIBRARY "" CACHE FILEPATH "staged motion-utils library")
set(OBJECTS_LIBRARY "" CACHE FILEPATH "staged objects marker library")
foreach(path IN ITEMS BVPC_COMMON_INCLUDE_DIR BVPC_COMMON_LIBRARY VELOCITY_SMOOTHER_LIBRARY VEHICLE_INFO_LIBRARY ROUTE_HANDLER_LIBRARY RTC_INCLUDE_DIR RTC_LIBRARY OBJECTS_INCLUDE_DIR OBJECTS_LIBRARY UTILS_INCLUDE_DIR UTILS_DEBUG_LIBRARY VEHICLE_INFO_INCLUDE_DIR MAP_MSGS_INCLUDE_DIR MOTION_UTILS_INCLUDE_DIR MOTION_UTILS_LIBRARY TRAJECTORY_INCLUDE_DIR PLANNING_FACTOR_INCLUDE_DIR)
  if(NOT EXISTS "${${path}}")
    message(FATAL_ERROR "Set ${path} to its staged ROS1 provider")
  endif()
endforeach()
catkin_package(INCLUDE_DIRS include LIBRARIES ${PROJECT_NAME} CATKIN_DEPENDS roscpp tier4_rtc_msgs unique_identifier_msgs)
include_directories(include ${BVPC_COMMON_INCLUDE_DIR} ${RTC_INCLUDE_DIR} ${OBJECTS_INCLUDE_DIR} ${UTILS_INCLUDE_DIR} ${VEHICLE_INFO_INCLUDE_DIR} ${MAP_MSGS_INCLUDE_DIR} ${MOTION_UTILS_INCLUDE_DIR} ${TRAJECTORY_INCLUDE_DIR} ${PLANNING_FACTOR_INCLUDE_DIR} ${EIGEN3_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS} ${catkin_INCLUDE_DIRS})
add_library(${PROJECT_NAME} SHARED
  src/scene_module_interface_with_rtc.cpp
  src/experimental/scene_module_interface_with_rtc.cpp)
target_link_libraries(${PROJECT_NAME} "${BVPC_COMMON_LIBRARY}" "${VELOCITY_SMOOTHER_LIBRARY}" "${VEHICLE_INFO_LIBRARY}" "${ROUTE_HANDLER_LIBRARY}" "${RTC_LIBRARY}" "${MOTION_UTILS_LIBRARY}" "${OBJECTS_LIBRARY}" "${UTILS_DEBUG_LIBRARY}" ${catkin_LIBRARIES})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_rtc_manager test/test_rtc_manager.cpp)
  if(TARGET test_rtc_manager)
    target_link_libraries(test_rtc_manager ${PROJECT_NAME} ${TRAJECTORY_LIBRARY} ${catkin_LIBRARIES})
    # Staged providers are raw shared libraries rather than imported CMake
    # targets; validate their full dependency closure at runtime instead of
    # forcing this unit-test executable to resolve every provider transitively.
    set_property(TARGET test_rtc_manager APPEND_STRING PROPERTY LINK_FLAGS " -Wl,--allow-shlib-undefined")
  endif()
endif()
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
