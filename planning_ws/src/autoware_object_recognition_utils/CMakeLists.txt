cmake_minimum_required(VERSION 3.0.2)
project(autoware_object_recognition_utils)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(catkin REQUIRED COMPONENTS autoware_interpolation autoware_perception_msgs autoware_utils_geometry autoware_utils_math geometry_msgs pcl_conversions pcl_ros roscpp sensor_msgs std_msgs tf2 tf2_eigen)
find_package(Boost REQUIRED)
find_package(PCL REQUIRED)
catkin_package(INCLUDE_DIRS include LIBRARIES ${PROJECT_NAME} CATKIN_DEPENDS autoware_interpolation autoware_perception_msgs autoware_utils_geometry autoware_utils_math geometry_msgs pcl_conversions pcl_ros roscpp sensor_msgs std_msgs tf2 tf2_eigen DEPENDS Boost PCL)
include_directories(include ${catkin_INCLUDE_DIRS} ${Boost_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
add_library(${PROJECT_NAME} SHARED src/conversion.cpp src/predicted_path_utils.cpp)
target_link_libraries(${PROJECT_NAME} ${catkin_LIBRARIES} ${PCL_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
 file(GLOB test_files test/src/*.cpp)
 catkin_add_gtest(test_${PROJECT_NAME} ${test_files})
 if(TARGET test_${PROJECT_NAME})
  target_link_libraries(test_${PROJECT_NAME} ${GTEST_MAIN_LIBRARIES} ${PROJECT_NAME} ${catkin_LIBRARIES} ${PCL_LIBRARIES})
 endif()
endif()
install(TARGETS ${PROJECT_NAME} ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
