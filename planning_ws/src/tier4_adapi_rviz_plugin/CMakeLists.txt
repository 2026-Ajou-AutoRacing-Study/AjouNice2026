cmake_minimum_required(VERSION 3.0.2)
project(tier4_adapi_rviz_plugin)
find_package(catkin REQUIRED COMPONENTS autoware_adapi_specs autoware_component_interface_utils autoware_vehicle_msgs geometry_msgs pluginlib roscpp rviz tier4_external_api_msgs tier4_planning_msgs)
find_package(Qt5 REQUIRED COMPONENTS Core Widgets)
set(CMAKE_AUTOMOC ON)
add_definitions(-DQT_NO_KEYWORDS)
catkin_package()
add_library(${PROJECT_NAME} SHARED src/route_tool.cpp src/route_panel.cpp src/operation_mode_debug_panel.cpp src/door_panel.cpp)
target_include_directories(${PROJECT_NAME} PRIVATE ${catkin_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME} ${catkin_LIBRARIES} ${rviz_DEFAULT_PLUGIN_LIBRARIES} Qt5::Widgets)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(FILES plugins/plugin_description.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/plugins)
install(DIRECTORY icons DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
