cmake_minimum_required(VERSION 3.14)
project(autoware_motion_velocity_obstacle_stop_module)
set(CMAKE_CXX_STANDARD 17)
find_package(catkin REQUIRED COMPONENTS autoware_internal_planning_msgs autoware_motion_utils autoware_motion_velocity_planner_common autoware_object_recognition_utils autoware_objects_of_interest_marker_interface autoware_perception_msgs autoware_planning_factor_interface autoware_planning_msgs autoware_signal_processing autoware_test_utils autoware_utils_debug autoware_utils_geometry autoware_utils_math autoware_utils_rclcpp autoware_utils_system autoware_utils_uuid autoware_utils_visualization autoware_vehicle_info_utils geometry_msgs grid_map_core pcl_conversions pluginlib roscpp tf2 tf2_ros visualization_msgs)
find_package(PCL REQUIRED COMPONENTS common filters kdtree segmentation)
catkin_package(LIBRARIES ${PROJECT_NAME} CATKIN_DEPENDS autoware_internal_planning_msgs autoware_motion_utils autoware_motion_velocity_planner_common autoware_object_recognition_utils autoware_objects_of_interest_marker_interface autoware_perception_msgs autoware_planning_factor_interface autoware_planning_msgs autoware_signal_processing autoware_test_utils autoware_utils_debug autoware_utils_geometry autoware_utils_math autoware_utils_rclcpp autoware_utils_system autoware_utils_uuid autoware_utils_visualization autoware_vehicle_info_utils geometry_msgs grid_map_core pcl_conversions pluginlib roscpp tf2 tf2_ros visualization_msgs)
add_library(${PROJECT_NAME} SHARED src/obstacle_stop_module.cpp)
target_include_directories(${PROJECT_NAME} BEFORE PUBLIC src ${catkin_INCLUDE_DIRS} ${PCL_INCLUDE_DIRS})
target_compile_options(${PROJECT_NAME} PRIVATE -fmax-errors=8)
target_link_libraries(${PROJECT_NAME} ${catkin_LIBRARIES} ${PCL_LIBRARIES})
target_link_options(${PROJECT_NAME} PRIVATE -Wl,--no-undefined)
install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY config DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(FILES plugins.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
