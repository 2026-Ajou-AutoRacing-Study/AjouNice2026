cmake_minimum_required(VERSION 3.0.2)
project(autoware_overlay_rviz_plugin)
find_package(catkin REQUIRED COMPONENTS autoware_internal_planning_msgs autoware_perception_msgs autoware_vehicle_msgs pluginlib roscpp roslib rviz rviz_2d_overlay_plugins rviz_2d_overlay_msgs)
find_package(Qt5 REQUIRED COMPONENTS Core Gui Widgets)
set(CMAKE_AUTOMOC ON)
set(CMAKE_INCLUDE_CURRENT_DIR ON)
add_definitions(-DQT_NO_KEYWORDS)
catkin_package(INCLUDE_DIRS include)
file(GLOB DISPLAY_HEADERS include/*.hpp)
file(GLOB DISPLAY_SOURCES src/*.cpp)
qt5_wrap_cpp(MOC_FILES include/signal_display.hpp)
add_library(${PROJECT_NAME} SHARED ${MOC_FILES} ${DISPLAY_HEADERS} ${DISPLAY_SOURCES})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
target_include_directories(${PROJECT_NAME} PUBLIC include ${catkin_INCLUDE_DIRS})
target_link_libraries(${PROJECT_NAME} ${catkin_LIBRARIES} Qt5::Core Qt5::Gui Qt5::Widgets)

install(TARGETS ${PROJECT_NAME} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_PACKAGE_INCLUDE_DESTINATION})
install(FILES plugins_description.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
install(DIRECTORY assets/ DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION}/assets)
