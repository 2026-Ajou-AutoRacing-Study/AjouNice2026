cmake_minimum_required(VERSION 3.0.2)
project(autoware_path_generator)
find_package(catkin REQUIRED COMPONENTS autoware_test_utils roslib autoware_motion_utils autoware_utils_debug autoware_utils_rclcpp autoware_utils_system autoware_utils_geometry autoware_utils_math autoware_vehicle_info_utils autoware_internal_debug_msgs autoware_vehicle_msgs roscpp geometry_msgs nav_msgs autoware_map_msgs autoware_planning_msgs autoware_internal_planning_msgs autoware_lanelet2_utils autoware_trajectory)
find_package(Eigen3 REQUIRED)
find_package(lanelet2_core REQUIRED)
set(Eigen3_INCLUDE_DIRS /usr/include/eigen3)
catkin_package(INCLUDE_DIRS include LIBRARIES ${PROJECT_NAME} CATKIN_DEPENDS autoware_motion_utils autoware_utils_debug autoware_utils_rclcpp autoware_utils_system autoware_utils_geometry autoware_utils_math autoware_vehicle_info_utils autoware_internal_debug_msgs autoware_vehicle_msgs roscpp geometry_msgs nav_msgs autoware_map_msgs autoware_planning_msgs autoware_internal_planning_msgs autoware_lanelet2_utils autoware_trajectory DEPENDS Eigen3 lanelet2_core)
include_directories(include ${catkin_INCLUDE_DIRS} ${EIGEN3_INCLUDE_DIRS})
# Original utils.cpp now supplies both border functions; retained legacy source
# remains recoverable but must not define the same ELF symbols twice.
add_library(${PROJECT_NAME} SHARED src/node.cpp src/utils.cpp src/route_path_generator_core.cpp src/route_path_generator_node.cpp)
target_link_libraries(${PROJECT_NAME} ${catkin_LIBRARIES} ${lanelet2_core_LIBRARIES})
target_compile_features(${PROJECT_NAME} PUBLIC cxx_std_17)
add_executable(route_path_generator_node src/route_path_generator_main.cpp)
target_link_libraries(route_path_generator_node ${PROJECT_NAME} ${catkin_LIBRARIES})
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_border_point test/test_border_point.cpp)
  if(TARGET test_border_point)
    target_link_libraries(test_border_point ${PROJECT_NAME} ${catkin_LIBRARIES})
  endif()
  catkin_add_gtest(test_route_path_generator_core test/test_route_path_generator_core.cpp)
  if(TARGET test_route_path_generator_core)
    target_compile_definitions(test_route_path_generator_core PRIVATE AW149_PATH_GENERATOR_TEST_MAP="${autoware_lanelet2_utils_DIR}/../sample_map/vm_03/left_hand/lanelet2_map.osm")
    target_link_libraries(test_route_path_generator_core ${PROJECT_NAME} ${catkin_LIBRARIES})
  endif()
  catkin_add_gtest(test_route_path_generator_node test/test_route_path_generator_node.cpp)
  if(TARGET test_route_path_generator_node)
    target_compile_definitions(test_route_path_generator_node PRIVATE AW149_PATH_GENERATOR_TEST_MAP="${autoware_lanelet2_utils_DIR}/../sample_map/vm_03/left_hand/lanelet2_map.osm")
    target_link_libraries(test_route_path_generator_node ${PROJECT_NAME} ${catkin_LIBRARIES})
  endif()
endif()
install(TARGETS ${PROJECT_NAME} route_path_generator_node
  LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION}
  RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/ DESTINATION include)
install(FILES package.xml DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})

target_link_options(${PROJECT_NAME} PRIVATE -Wl,--no-undefined)
add_executable(path_generator_node src/full_path_generator_main.cpp)
target_link_libraries(path_generator_node ${PROJECT_NAME} ${catkin_LIBRARIES})
install(TARGETS path_generator_node RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY config param test_route DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})

if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_full_original_geometry test/test_lanelet.cpp test/test_path_cut.cpp test/test_goal_connection.cpp test/test_turn_signal.cpp test/test_current_dense_centerline.cpp test/test_full_native_node.cpp test/full_test_main.cpp)
  if(TARGET test_full_original_geometry)
    target_include_directories(test_full_original_geometry BEFORE PRIVATE include src ${catkin_INCLUDE_DIRS})
    target_link_libraries(test_full_original_geometry ${PROJECT_NAME} ${catkin_LIBRARIES})
  endif()
endif()
