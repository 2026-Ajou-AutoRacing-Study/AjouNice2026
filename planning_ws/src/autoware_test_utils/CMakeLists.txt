cmake_minimum_required(VERSION 3.0.2)
project(autoware_test_utils)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
find_package(catkin REQUIRED COMPONENTS
  autoware_adapi_v1_msgs autoware_internal_planning_msgs autoware_lanelet2_extension autoware_map_msgs autoware_perception_msgs autoware_planning_msgs autoware_utils_geometry geometry_msgs lanelet2_io nav_msgs roscpp roslib rosgraph_msgs std_msgs std_srvs tf2 tf2_geometry_msgs tf2_ros unique_identifier_msgs)
find_package(yaml-cpp REQUIRED)
set(yaml-cpp_LIBRARIES yaml-cpp)
catkin_package(INCLUDE_DIRS include LIBRARIES autoware_test_utils CATKIN_DEPENDS autoware_adapi_v1_msgs autoware_internal_planning_msgs autoware_lanelet2_extension autoware_map_msgs autoware_perception_msgs autoware_planning_msgs autoware_utils_geometry geometry_msgs lanelet2_io nav_msgs roscpp roslib rosgraph_msgs std_msgs std_srvs tf2 tf2_geometry_msgs tf2_ros unique_identifier_msgs DEPENDS yaml-cpp)
include_directories(include ${catkin_INCLUDE_DIRS})
add_library(autoware_test_utils SHARED src/autoware_test_utils.cpp src/mock_data_parser.cpp)
target_link_libraries(autoware_test_utils ${catkin_LIBRARIES} yaml-cpp)
add_executable(topic_snapshot_saver src/topic_snapshot_saver.cpp)
target_link_libraries(topic_snapshot_saver autoware_test_utils ${catkin_LIBRARIES} yaml-cpp)
if(CATKIN_ENABLE_TESTING)
  catkin_add_gtest(test_mock_data_parser test/test_mock_data_parser.cpp)
  if(TARGET test_mock_data_parser)
    target_link_libraries(test_mock_data_parser autoware_test_utils ${catkin_LIBRARIES} yaml-cpp)
  endif()
  catkin_add_gtest(test_autoware_test_manager test/test_autoware_test_manager.cpp)
  if(TARGET test_autoware_test_manager)
    target_link_libraries(test_autoware_test_manager autoware_test_utils ${catkin_LIBRARIES} yaml-cpp)
  endif()
endif()
install(TARGETS autoware_test_utils topic_snapshot_saver ARCHIVE DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} LIBRARY DESTINATION ${CATKIN_PACKAGE_LIB_DESTINATION} RUNTIME DESTINATION ${CATKIN_PACKAGE_BIN_DESTINATION})
install(DIRECTORY include/ DESTINATION ${CATKIN_GLOBAL_INCLUDE_DESTINATION})
install(DIRECTORY config test_map test_data rviz DESTINATION ${CATKIN_PACKAGE_SHARE_DESTINATION})
